--- COMMAND 7: Add new URL routes ---
python -c "
code = open('/home/runner/workspace/core/urls.py').read()
routes = [
    \"    path('stock-log/', views.stock_movement_log, name='stock_movement_log'),\",
    \"    path('director/all-stock/', views.director_all_branch_stock, name='director_all_branch_stock'),\",
    \"    path('director/all-activities/', views.director_all_activities, name='director_all_activities'),\",
    \"    path('send-daily-report/', views.send_daily_report_whatsapp, name='send_daily_report_whatsapp'),\",
    \"    path('send-stock-request/', views.send_stock_request_whatsapp, name='send_stock_request_whatsapp'),\",
]
for r in routes:
    name = r.split(\"name='\")[1].rstrip(\"'),\").strip()
    if name not in code:
        code = code.rstrip().rstrip(']').rstrip() + '\n' + r + '\n]\n'
        print('Added:', name)
    else:
        print('Already exists:', name)
open('/home/runner/workspace/core/urls.py','w').write(code)
print('Done')
"
