python manage.py makemigrations

🔷 STEP 2 — MODIFY CHECK-IN LOGIC (Flat ₦250)

Replace your lateness calculation with:

from datetime import time, datetime
from django.utils import timezone
from decimal import Decimal

now = timezone.localtime()
official_time = datetime.combine(now.date(), time(8,0))
official_time = timezone.make_aware(official_time)

if now > official_time:
    attendance.is_late = True
    attendance.deduction_amount = Decimal('250.00')
    attendance.save()

✔ No minute calculations
✔ Every late = ₦250
✔ Clean and simple

🔷 STEP 4 — PDF ATTENDANCE EXPORT

Install:

pip install reportlab
Create PDF Export View
from reportlab.platypus import SimpleDocTemplate, Table, TableStyle, Paragraph
from reportlab.lib import colors
from reportlab.lib.styles import getSampleStyleSheet
from reportlab.lib import colors
from django.http import HttpResponse
from django.utils import timezone
from django.db.models import Sum
import calendar
from io import BytesIO

def export_monthly_attendance_pdf(request):

    today = timezone.now()
    month = today.month
    year = today.year

    records = Attendance.objects.filter(
        date__month=month,
        date__year=year
    )

    buffer = BytesIO()
    doc = SimpleDocTemplate(buffer)
    elements = []

    data = [["Staff", "Branch", "Date", "Late", "Deduction"]]

    for record in records:
        data.append([
            record.user.username,
            record.branch.name,
            str(record.date),
            "Yes" if record.is_late else "No",
            f"₦{record.deduction_amount}"
        ])

    table = Table(data)
    table.setStyle(TableStyle([
        ('BACKGROUND',(0,0),(-1,0),colors.grey),
        ('GRID',(0,0),(-1,-1),1,colors.black),
    ]))

    elements.append(table)
    doc.build(elements)

    buffer.seek(0)

    return HttpResponse(
        buffer,
        content_type='application/pdf',
        headers={'Content-Disposition': 'attachment; filename="Monthly_Attendance_Report.pdf"'}
    )

Now Director can download monthly report.

🔷 STEP 5 — AUTOMATIC MONTH-END EMAIL TO DIRECTOR

We schedule this using cron or Celery.

Email Function
from django.core.mail import EmailMessage
from django.conf import settings

def send_monthly_attendance_email():

    today = timezone.now()
    last_day = calendar.monthrange(today.year, today.month)[1]

    if today.day != last_day:
        return

    records = Attendance.objects.filter(
        date__month=today.month,
        date__year=today.year
    )

    total_deductions = records.aggregate(
        Sum("deduction_amount")
    )["deduction_amount__sum"] or 0

    subject = f"Monthly Attendance Report - {today.strftime('%B %Y')}"
    message = f"""
    Total Lateness Deductions: ₦{total_deductions}
    Total Records: {records.count()}
    """

    email = EmailMessage(
        subject,
        message,
        settings.DEFAULT_FROM_EMAIL,
        ["director@email.com"]
    )

    email.send()
